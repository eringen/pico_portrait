cmake_minimum_required(VERSION 3.20)
if(NOT PICO_SDK_PATH AND DEFINED ENV{PICO_SDK_PATH})
  set(PICO_SDK_PATH "$ENV{PICO_SDK_PATH}")
endif()
if(NOT PICO_SDK_PATH)
  message(FATAL_ERROR "Set PICO_SDK_PATH to the pinned SDK; see docs/build.md")
endif()
set(PICO_BOARD pico_w CACHE STRING "Target board")
if(NOT PICO_BOARD STREQUAL "pico_w")
  message(FATAL_ERROR "This diagnostic is for RP2040 Pico W only")
endif()
include(${PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(portrait_diagnostic C CXX ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
option(PORTRAIT_USB_HOST "Start host controller: ONLY after VBUS wiring is verified" OFF)
set(PORTRAIT_CONFIG_DIR "${CMAKE_CURRENT_LIST_DIR}" CACHE PATH "Directory containing optional config.local.h")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_LIST_DIR}/../scripts/check_sdk.py ${PICO_SDK_PATH}
  COMMAND_ERROR_IS_FATAL ANY)
pico_sdk_init()
# The SDK permits dependency overrides; reject them here so the checked pins
# describe the libraries actually linked by this project.
foreach(pair "PICO_TINYUSB_PATH|tinyusb" "PICO_CYW43_DRIVER_PATH|cyw43-driver" "PICO_LWIP_PATH|lwip")
  string(REPLACE "|" ";" parts "${pair}")
  list(GET parts 0 variable)
  list(GET parts 1 directory)
  file(REAL_PATH "${${variable}}" actual)
  file(REAL_PATH "${PICO_SDK_PATH}/lib/${directory}" expected)
  if(NOT actual STREQUAL expected)
    message(FATAL_ERROR "${variable} must use the checked SDK submodule")
  endif()
endforeach()
add_executable(portrait_diagnostic src/main.c src/usb_diag.c src/diagnostic.c src/http.c src/oled.c)
target_include_directories(portrait_diagnostic PRIVATE src "${PORTRAIT_CONFIG_DIR}" ${CMAKE_CURRENT_LIST_DIR})
target_compile_definitions(portrait_diagnostic PRIVATE
  PORTRAIT_USB_HOST=$<BOOL:${PORTRAIT_USB_HOST}>
  PICO_STACK_SIZE=0x1000
  PICO_DEFAULT_UART=1 PICO_DEFAULT_UART_TX_PIN=4 PICO_DEFAULT_UART_RX_PIN=5)
target_compile_options(portrait_diagnostic PRIVATE -Wall -Wextra -fstack-usage)
target_link_libraries(portrait_diagnostic pico_stdlib hardware_i2c tinyusb_host pico_cyw43_arch_lwip_poll)
pico_enable_stdio_usb(portrait_diagnostic 0)
pico_enable_stdio_uart(portrait_diagnostic 1)
pico_set_program_name(portrait_diagnostic "Portrait no-motion diagnostic")
pico_set_program_version(portrait_diagnostic "0.2.0")
pico_add_extra_outputs(portrait_diagnostic)
